\subsection*{الف}
چون سیستم علّی و حقیقی است، تبدیل هیلبرت بین بخش‌های حقیقی و موهومی برقرار است و تابع موهومی فرد خواهد بود. با توجه به شکل استاندارد
\begin{flalign*}
H(j\omega)=\frac{1}{1+j\omega}
= \frac{1-j\omega}{1+\omega^2}
\quad \Longrightarrow \quad
H_R(j\omega)=\frac{1}{1+\omega^2},\;
H_I(j\omega)=-\frac{\omega}{1+\omega^2}. &&
\end{flalign*}
پس نتیجه انتگرال هیلبرت نیز
\begin{flalign*}
H_I(j\omega)=-\frac{\omega}{1+\omega^2} &&
\end{flalign*}
می‌باشد.

\subsection*{ب}
اکنون
\begin{flalign*}
H(j\omega)=H_R(j\omega)+jH_I(j\omega)
=\frac{1-j\omega}{1+\omega^2}
=\frac{1}{1+j\omega}. &&
\end{flalign*}
بنابراین با تبدیل فوریه معکوس داریم
\begin{flalign*}
h(t)=e^{-t}u(t). &&
\end{flalign*}

\subsection*{ج}
سیستم پایدار است زیرا
\begin{flalign*}
\int_{-\infty}^{\infty} |h(t)|\,dt
=\int_{0}^{\infty} e^{-t}dt
=1<\infty. &&
\end{flalign*}
