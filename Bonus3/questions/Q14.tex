\textbf{2.77.} Let $x[n]$ and $y[n]$ denote complex sequences and $X(e^{j\omega})$ and $Y(e^{j\omega})$ their respective Fourier transforms.
\begin{enumerate}
    \item[\textbf{(a)}] By using the convolution theorem (Theorem 6 in Table 2.2) and appropriate properties from Table 2.2, determine, in terms of $x[n]$ and $y[n]$, the sequence whose Fourier transform is $X(e^{j\omega})Y^*(e^{j\omega})$.
    \item[\textbf{(b)}] Using the result in Part (a), show that
    \begin{equation}\tag{P2.77-1}
        \sum_{n=-\infty}^{\infty} x[n]y^*[n] = \frac{1}{2\pi} \int_{-\pi}^{\pi} X(e^{j\omega})Y^*(e^{j\omega})d\omega.
    \end{equation}
\end{enumerate}
